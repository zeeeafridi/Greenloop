function saveRecord() { const date = document.getElementById("date").value; const production = Number(document.getElementById("productionQty").value); const exportQty = Number(document.getElementById("exportQty").value) || 0; if (!date || Number.isNaN(production)) { alert("Please enter Date and Produced Quantity."); return; } const previousStock = getPreviousStock(); const record = { date: date, model: document.getElementById("model").value, storage: document.getElementById("storage").value, color: document.getElementById("color").value, grade: document.getElementById("grade").value, production: production, exportQty: exportQty, openingStock: previousStock, closingStock: previousStock + production - exportQty }; const records = getRecords(); const existingIndex = records.findIndex(function(item) { return item.date === record.date && item.model === record.model && item.storage === record.storage && item.color === record.color && item.grade === record.grade; }); if (existingIndex >= 0) { const oldRecord = records[existingIndex]; const totalProduction = (Number(oldRecord.production) || Number(oldRecord.quantity) || 0) + production; const totalExport = (Number(oldRecord.exportQty) || 0) + exportQty; const openingStock = Number(oldRecord.openingStock) || previousStock; records[existingIndex] = { ...record, production: totalProduction, exportQty: totalExport, openingStock: openingStock, closingStock: openingStock + totalProduction - totalExport }; } else { if (record.closingStock < 0) { alert("Export quantity is more than available stock."); return; } records.push(record); } localStorage.setItem("productionRecords", JSON.stringify(records)); document.getElementById("productionQty").value = ""; document.getElementById("exportQty").value = "0"; showRecords(); updateStockPreview(); }
